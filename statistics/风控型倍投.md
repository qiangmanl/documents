from matplotlib import pyplot as plt
import random

class trouble_trade:
    def __init__(self,
        times = 3,
        balance = 10000,
        commission=0.007,
        hold = 0.6,
        limited = 5,
        begin = 8,
        long = 1,
        init_order = 1,
        size = 365*24*60
    ):
        
        self.times = times
        self.balance = balance
        self.commission = commission
        self.hold = hold
        self.limited = limited
        self.begin = begin
        self.long = long
        self.init_order = init_order
        self.hold_balance = balance * hold
        self.size = size
        self.history = []
        self.end = False
        
    def get_sample(self):
        return random.choices([0,1],k=self.size)


    def run(self):
        self.remembers = []
        samples = self.get_sample()
        lock = 0
        self.order = self.init_order
        for i in range(len(samples)):
            self.history.append(self.balance)
            if samples[i] == self.long:
                self.remembers.append(self.long)
                if len(self.remembers) > self.begin:
                    self.balance -= self.order
                    if lock == self.limited:
                        self.order = self.init_order 
                        self.remembers = []
                        lock = 0
                    else:
                        if self.balance - self.order * self.times  >  self.hold_balance:
                            self.order *= self.times 
                            lock += 1

                        else:
                            break
                    
            else:
                if len(self.remembers) > self.begin:
                    self.balance += self.order * (1 - self.commission)
                    self.order = self.init_order
                    lock = 0
                self.remembers = []     
        self.end = True
        return self

    def show_history(self):
        if self.end:
            plt.plot(self.history)
            plt.show()

        self.loss_limit = loss_limit
        self.order_times = order_times
        self.times = times
        self.end = False
        self.history = []

    def update_returns(self, current):
        self.returns += self.get_return(current) * self.order_times - self.commission

    def _init_order(self):
        self.order_times = 1
        self.remembers = []
        self.lock = 0

    def _keep_order(self):
        self.order_times *= self.times 
        self.lock += 1

    def get_return(self, current):
        return self.samples.values[current]

    def _keep_trade(self,current):
        if self.get_return(current) < 0:
            self.remembers.append(1)
            if len(self.remembers) > self.begin:
                self.update_returns(current)
                if self.lock == self.limit_step:
                    self._init_order()
                else:
                    if self.returns < self.loss_limit: 
                        return False

                    else:
                        self._keep_order()
        else:
            if len(self.remembers) > self.begin:
                self.update_returns(current)
            self._init_order()  
        return True
