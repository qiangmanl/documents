#### 链接地址：https://q.stock.sohu.com/cn/bk_4539.shtml
```python


import requests
from lxml import html

#新能源车概念
url = "https://q.stock.sohu.com/cn/bk_4539.shtml"

response = requests.get(url)
html_content = response.text

tree = html.fromstring(html_content)
i = 1
data = {
    "新能源车" : []
}
while True:
    result = tree.xpath(f'/html/body/div[4]/div/div[3]/table/tbody/tr[{i}]/td[1]')
    if result:
        # 输出第一个匹配的元素的文本内容
        data["新能源车"].append(result[0].text_content())
    else:
        break
    i += 1
 
```
