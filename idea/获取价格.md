import pandas as pd
import cryptocompare

# 设置参数
symbol = 'BTC'
currency = 'USD'
limit = 2000  # 获取最近60个小时的数据，你可以根据需要调整

# 获取历史数据
historical_data = cryptocompare.get_historical_price_hour(symbol, currency, limit=limit)

# 转换为Pandas DataFrame
df = pd.DataFrame(historical_data)
df['time'] = pd.to_datetime(df['time'], unit='s')  # 将时间戳转换为可读格式

# 打印数据
