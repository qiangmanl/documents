```python
import pandas as pd
import numpy as np
from datetime import datetime, timedelta

start_time = datetime(2023, 1, 1, 0, 0)  # 2023年1月1日上午9点
end_time = datetime(2023, 8, 1, 23, 59)   # 2023年1月1日下午4点
interval = timedelta(minutes=1)
time_index = pd.date_range(start=start_time, end=end_time, freq=interval)
simulated_data = range(1,len(time_index) + 1)  # 1440分钟对应一天
df = pd.DataFrame(data={'Value': simulated_data}, index=time_index)
print(df)

```
#### 按日期
```python
# 按照 时间单位 计算后整合成数据  range(1,len(time_index) + 1) #一小时 60-1 = 59 没毛病
df.resample('1H').apply(lambda x: x.iloc[-1] - x.iloc[0])

```


#### 按一天分割成N组
```python
daily_groups = df.groupby(pd.Grouper(freq='D'))
for _index, daily_data in daily_groups:
    print(daily_data)
    
    
```

####  将一个小时按分钟数分割成2份
```python
def split_data(hour):
    #不要大于 24
    return daily_data[:hour * 60],daily_data[hour * 60:]
```
